```java
package org.modelio.platform.query.internal;

import java.text.MessageFormat;
import org.modelio.platform.search.engine.plugin.SearchEngine;
import org.modelio.platform.utils.i18n.BundledMessages;

/**
 * Null-safe I18n helper for the internal query package.
 * Falls back to the raw key (or MessageFormat formatting) when
 * {@link SearchEngine#I18N} is null (e.g. in unit-test context where
 * the OSGi bundle activator never runs).
 */
public final class SearchEngineI18N {

    private SearchEngineI18N() {
        // utility class
    }

    /**
     * Returns the localized message for the given key,
     * or a safe fallback when I18N is unavailable.
     *
     * @param key  the message key (e.g. {@code "MQL.resolver.kind.unknown"})
     * @param args optional MessageFormat arguments
     * @return localized message, or key / formatted key as fallback
     */
    public static String msg(String key, Object... args) {
        BundledMessages i18n = SearchEngine.I18N;
        if (i18n != null) {
            return i18n.getMessage(key, args);
        }
        // Fallback for test context: use MessageFormat if args provided
        if (args.length == 0) {
            return key;
        }
        return MessageFormat.format(key, args);
    }

}
```

No imports of SearchEngine or BundledMessages should be added to other files yet — that is Plan 07-02.
