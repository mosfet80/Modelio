// -----------------------------------------------------------------------
// Test 5b: emitAllDepths=false with maxDepth truncation
// -----------------------------------------------------------------------
@Test
void repeat_emitAllDepthsFalse_maxDepthTruncation() {
    QueryEngine engine = createEngine();

    // root → pkgA → pkgB → leaf
    // maxDepth=2, emitAllDepths=false:
    //   depth 1 → discovers pkgA
    //   depth 2 → discovers pkgB (maxDepth reached, stop)
    // Only the maxDepth frontier (pkgB) should be emitted, not pkgA.
    RepeatStep repeatStep = new RepeatStep(List.of(new TraverseStep("OwnedElement")));
    repeatStep.setEmitAllDepths(false);

    Query query = new Query(
            new AllSource("Package"),
            List.of(nameFilter("root"), repeatStep),
            new LimitsSpec(2, null, null));

    IQueryResponse response = engine.syncExec(query);

    assertThat(response.items())
            .extracting(NodeWithExplainOut::node)
            .containsExactly(this.pkgB)
            .doesNotContain(this.root, this.pkgA, this.leaf);
    assertThat(response.state()).isCompletedWithValue(QueryCompletionStatus.ENDED);
}
```

This tests that when maxDepth caps the traversal and emitAllDepths=false, only the frontier at the
maxDepth level is emitted, not intermediate levels.

// -----------------------------------------------------------------------
// Test 5c: emitAllDepths=false — second repeat sees all nodes already visited
// -----------------------------------------------------------------------
@Test
void repeat_emitAllDepthsFalse_globalVisitedFiltersAll() {
    QueryEngine engine = createEngine();

    // Two consecutive RepeatSteps sharing globalVisited.
    // First repeat: root → pkgA → pkgB → leaf, emitAllDepths=false → emits leaf
    // Second repeat: same source (root), same body — but pkgA, pkgB, leaf are
    // already in globalVisited. At depth 1, body returns pkgA which is already
    // visited → nextFrontier is empty → depth==1 means depth>1 guard prevents
    // emission → second repeat produces nothing.
    // Net result: only leaf (from the first repeat).
    RepeatStep repeat1 = new RepeatStep(List.of(new TraverseStep("OwnedElement")));
    repeat1.setEmitAllDepths(false);

    RepeatStep repeat2 = new RepeatStep(List.of(new TraverseStep("OwnedElement")));
    repeat2.setEmitAllDepths(false);

    Query query = new Query(
            new AllSource("Package"),
            List.of(nameFilter("root"), repeat1, repeat2),
            new LimitsSpec());

    IQueryResponse response = engine.syncExec(query);

    // Only leaf from the first repeat; second repeat produces nothing
    assertThat(response.items())
            .extracting(NodeWithExplainOut::node)
            .containsExactly(this.leaf);
    assertThat(response.state()).isCompletedWithValue(QueryCompletionStatus.ENDED);
}
```

This tests Finding 1's accepted behaviour: when a second RepeatStep in the same pipeline
